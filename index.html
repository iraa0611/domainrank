import { createClient } from "npm:@supabase/supabase-js@2"; Deno.serve(async (req) => { try { if (req.method !== "POST") { return new Response( JSON.stringify({ error: "POST required" }), { status: 405, headers: { "Content-Type": "application/json" }, } ); } const body = await req.json(); const domainId = Number(body.domain_id); const bidderName = String(body.bidder_name || "").trim(); const bidAmount = Number(body.bid_amount); if (!domainId || !bidderName || !Number.isFinite(bidAmount)) { return new Response( JSON.stringify({ error: "domain_id, bidder_name and bid_amount are required" }), { status: 400, headers: { "Content-Type": "application/json" }, } ); } if (bidAmount <= 0) { return new Response( JSON.stringify({ error: "Bid must be greater than zero" }), { status: 400, headers: { "Content-Type": "application/json" }, } ); } const supabaseUrl = Deno.env.get("SUPABASE_URL")!; const serviceRoleKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!; const supabase = createClient( supabaseUrl, serviceRoleKey ); const { data: domain, error: domainError } = await supabase .from("domains") .select("id, domain, current_bid") .eq("id", domainId) .single(); if (domainError || !domain) { return new Response( JSON.stringify({ error: "Domain not found" }), { status: 404, headers: { "Content-Type": "application/json" }, } ); } if (bidAmount <= Number(domain.current_bid)) { return new Response( JSON.stringify({ error: `Bid must be higher than $${domain.current_bid}` }), { status: 400, headers: { "Content-Type": "application/json" }, } ); } const { error: bidError } = await supabase .from("bids") .insert({ domain_id: domain.id, bidder_name: bidderName, bid_amount: bidAmount, payment_status: "pending" }); if (bidError) { return new Response( JSON.stringify({ error: bidError.message }), { status: 500, headers: { "Content-Type": "application/json" }, } ); } return new Response( JSON.stringify({ success: true, message: "Bid created as pending", domain: domain.domain, bid_amount: bidAmount }), { status: 200, headers: { "Content-Type": "application/json" }, } ); } catch (error) { return new Response( JSON.stringify({ error: String(error) }), { status: 500, headers: { "Content-Type": "application/json" }, } ); } });
